Turtle config

Paste this shape into `/fleet/config.lua`, replacing `HOST` when Minecraft runs on another machine.

return {
  turtle_id = "fleet-dev-001",
  fleet_url = "ws://HOST:8787/turtle/ws",
  pairing_token = "dev-pairing-token",
  runtime_version = "0.1.0",
  dimension = "overworld",
  initial_facing = "north",
  initial_position = nil
}
